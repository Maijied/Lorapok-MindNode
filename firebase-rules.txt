# Firestore Security Rules

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    match /notes/{noteId} {
      // Since we use a Key-based system without traditional Auth,
      // any client with the NoteId can read the encrypted blob.
      // The security is provided by the AES-GCM encryption (E2EE).
      allow read, write: if true;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
